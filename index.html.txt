<script>

const modal =
    document.getElementById("modal");

const toast =
    document.getElementById("toast");

const title =
    document.getElementById("modal-title");

const text =
    document.getElementById("modal-text");
const input =
    document.getElementById("user-input");

const submit =
    document.getElementById("modal-submit");


/* باز کردن پنجره ورود */

function openModal(
    kind = "login",
    plan = ""
) {

    modal.classList.add("show");

    if (kind === "buy") {

        title.textContent =
            "ثبت سفارش چالش";

        text.textContent =
            "پلن انتخابی: " +
            plan +
            " — اطلاعات تماس خود را وارد کنید.";

        submit.textContent =
            "ادامه سفارش";

    } else {

        title.textContent =
            "ورود به پنل";

        text.textContent =
            "شماره موبایل یا ایمیل خود را وارد کنید.";

        submit.textContent =
            "ادامه";
    }

    setTimeout(() => {

        input.focus();

    }, 100);
}


/* دکمه ورود */

document
    .querySelectorAll("[data-modal]")
    .forEach(button => {

        button.addEventListener(
            "click",
            () => {

                openModal(
                    button.dataset.modal
                );

            }
        );

    });


/* دکمه‌های انتخاب چالش */

document
    .querySelectorAll(".buy")
    .forEach(button => {

        button.addEventListener(
            "click",
            () => {

                openModal(
                    "buy",
                    button.dataset.plan
                );

            }
        );

    });


/* بستن Modal */

document
    .querySelector(".close")
    .onclick = () => {

        modal.classList.remove("show");

    };


/* کلیک بیرون از Modal */

modal.addEventListener(
    "click",
    event => {

        if (event.target === modal) {

            modal.classList.remove("show");

        }

    }
);


/* ثبت فرم */

submit.onclick = () => {

    if (!input.value.trim()) {

        input.focus();

        return;
    }

    modal.classList.remove("show");

    toast.textContent =
        "درخواست شما ثبت شد.";

    toast.classList.add("show");


    setTimeout(() => {

        toast.classList.remove("show");

    }, 3000);

};

</script>
